prot_hidro.ino — PROGRAM ESP32 (FLOOD & DRAIN)

/* ============================================================
   PROT-HIDRO v2.0 — Hidroponik FLOOD & DRAIN + Penghalau Hama
   Board : ESP32 DevKit V1
   ------------------------------------------------------------
   - Pompa mengisi bak tanam; sensor level di dinding bak:
     level < 30% -> relay ON (mengisi) | > 70% -> OFF (cukup)
     | pengaman: isi maksimal 15 detik lalu berhenti
   - PIR gerak DAN objek < 60 cm -> sirene 8 detik
   Wiring penting:
   - ECHO HC-SR04 WAJIB lewat pembagi R1 1k / R2 2k (5V -> 3.3V)
   - Buzzer digerakkan TRANSISTOR Q1 (basis via R3 1k)
   - D1 1N4007 paralel pompa (katoda/strip ke +)
   - C1 220uF di kutub adapter agar ESP32 tidak brown-out
   - Semua pin "5V/GND" pada skema = rel daya adapter
   - Kebanyakan modul relay aktif-LOW
   ============================================================ */

#define PIN_LEVEL   34   // AO sensor level kapasitif (ADC1, input-only)
#define PIN_PIR     27   // OUT sensor PIR
#define PIN_TRIG    5    // Trig HC-SR04
#define PIN_ECHO    18   // Echo (lewat pembagi R1 1k + R2 2k!)
#define PIN_RELAY   26   // IN modul relay (aktif-LOW)
#define PIN_BUZZER  25   // ke BASIS Q1 via R3 1k (bukan buzzer langsung!)

#define BACA_BASAH   1400   // ADC saat air MENYENTUH sensor -> kalibrasi!
#define BACA_KERING  3100   // ADC saat air DI BAWAH sensor  -> kalibrasi!
#define RELAY_ON     LOW
#define RELAY_OFF    HIGH
#define AMBANG_ON     30        // % ; di bawah ini bak diisi
#define AMBANG_OFF    70        // % ; di atas ini pompa berhenti
#define JARAK_ANCAMAN 60        // cm
#define DURASI_ALARM  8000UL
#define LAMA_ISI_MAX  15000UL  // pengaman: bocor / sensor mati

int  levelPersen = 0;
bool pompaHidup  = false;
bool alarmAktif  = false;
bool nadaTinggi  = false;
unsigned long tAlarm, tIsi, tSensor, tBuzz, tCetak;

int bacaLevelPersen() {
  int adc = analogRead(PIN_LEVEL);
  return constrain(map(adc, BACA_KERING, BACA_BASAH, 0, 100), 0, 100);
}

float bacaJarakCm() {
  digitalWrite(PIN_TRIG, LOW);  delayMicroseconds(3);
  digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  long durasi = pulseIn(PIN_ECHO, HIGH, 30000UL);
  if (durasi == 0) return 999.0;
  return durasi / 58.0;
}

void sireneTick() {            // non-blocking, via transistor Q1
  if (millis() - tBuzz < 140) return;
  tBuzz = millis();
  nadaTinggi = !nadaTinggi;
  tone(PIN_BUZZER, nadaTinggi ? 980 : 720);
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TRIG, OUTPUT);  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_PIR, INPUT);    pinMode(PIN_RELAY, OUTPUT);
  pinMode(PIN_BUZZER, OUTPUT);
  digitalWrite(PIN_RELAY, RELAY_OFF);  // fail-safe: pompa mati saat boot
  noTone(PIN_BUZZER);
  Serial.println(F("PROT-HIDRO v2 (flood & drain) siap."));
}

void loop() {
  unsigned long kini = millis();

  if (kini - tSensor >= 200) {
    tSensor = kini;
    levelPersen = bacaLevelPersen();
    float cm = bacaJarakCm();
    bool gerak = (digitalRead(PIN_PIR) == HIGH);

    /* ---------- penghalang perimeter ---------- */
    if (!alarmAktif && gerak && cm < JARAK_ANCAMAN) {
      alarmAktif = true;  tAlarm = kini;
      Serial.print(F("[ALARM] Penyusup! "));
      Serial.print(cm, 0); Serial.println(F(" cm."));
    }

    /* ---------- flood & drain (histeresis) ---------- */
    if (!alarmAktif) {
      if (!pompaHidup && levelPersen < AMBANG_ON) {
        pompaHidup = true;  tIsi = kini;
        Serial.println(F("[ISI] Level surut -> pompa ON"));
      }
      if (pompaHidup && (levelPersen > AMBANG_OFF ||
          kini - tIsi > LAMA_ISI_MAX)) {
        pompaHidup = false;
        Serial.println(F("[ISI] Bak cukup -> pompa OFF"));
      }
    }
  }

  digitalWrite(PIN_RELAY, pompaHidup ? RELAY_ON : RELAY_OFF);

  if (alarmAktif) {
    sireneTick();
    if (millis() - tAlarm > DURASI_ALARM) {
      alarmAktif = false;  noTone(PIN_BUZZER);
      Serial.println(F("[ALARM] Perimeter aman kembali."));
    }
  }

  if (millis() - tCetak >= 1000) {
    tCetak = millis();
    Serial.print(F("level="));  Serial.print(levelPersen);
    Serial.print(F("% pompa=")); Serial.print(pompaHidup);
    Serial.print(F(" alarm="));  Serial.println(alarmAktif);
  }
}

ANOTASI A-01 · KALIBRASI LEVEL

  • A1Serial Monitor: catat ADC saat air menyentuh sensor (→ BACA_BASAH) dan saat surut (→ BACA_KERING).
  • A2Sensor kapasitif ditempel dinding dalam bak setinggi LEVEL TARGET, dibungkus mika tipis — terbaca, bebas korosi.

ANOTASI A-02 · RESISTOR

  • B1R1 1k + R2 2k pembagi ECHO (5V→3,3V) — WAJIB; titik tengahnya ke GPIO18.
  • B2R3 1k pembatas arus basis Q1; arus buzzer dipikul transistor, bukan GPIO25.

ANOTASI A-03 · DAYA & FAIL-SAFE

  • C1Pompa di kontak NO relay aktif-LOW → default MATI; hang/lepas kabel = berhenti mengisi, bukan meluap.
  • C2D1 melindungi kontak relay; C1 220µF mencegah brown-out. Semua GND = satu kesatuan.
